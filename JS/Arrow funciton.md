Normal function:
```js
function greetings(name) {
  console.log("Hello, " + name + "!");
}
```
arrow funciton:
```js
const greetings = (name) => {
  console.log("Hello, " + name + "!");
};

// if one parmaeter only:
const greetings = name => {
  console.log("Hello, " + name + "!");
};

// if no parameters:
const greetings = () => {
  console.log("Hello");
};

```
